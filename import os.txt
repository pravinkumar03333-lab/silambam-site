def is_tamil(text):

    for char in text:

        if "\u0B80" <= char <= "\u0BFF":
            return True

    return False


def detect_language(text):

    if is_tamil(text):
        return "ta"

    return "en"

tts = pyttsx3.init()

tts.setProperty(
    "rate",
    160
)

tts.setProperty(
    "volume",
    1.0
)


def get_voices():

    try:
        return tts.getProperty("voices")
    except Exception:
        return []


def find_voice(language):

    voices = get_voices()

    if language == "ta":

        keywords = [
            "tamil",
            "ta-in",
            "ta_in",
            "tamil india"
        ]

    else:

        keywords = [
            "english",
            "en-in",
            "en-us",
            "en-gb",
            "india"
        ]

    for voice in voices:

        information = (
            str(voice.id)
            + " "
            + str(voice.name)
            + " "
            + str(
                getattr(
                    voice,
                    "languages",
                    ""
                )
            )
        ).lower()

        for keyword in keywords:

            if keyword in information:
                return voice.id

    return None


def speak_text(text, language):

    global speaking
    global talk_frame

    try:

        voice_id = find_voice(
            language
        )

        if voice_id:

            tts.setProperty(
                "voice",
                voice_id
            )

        else:

            if language == "ta":

                print(
                    "Tamil TTS voice was not found."
                )

                print(
                    "Install a Tamil Windows voice."
                )

        speaking = True
        talk_frame = 0

        tts.say(text)
        tts.runAndWait()

    except Exception as error:

        print(
            "TTS error:",
            error
        )

    finally:

        speaking = False
        talk_frame = 0


def speak_async(text, language):

    threading.Thread(
        target=speak_text,
        args=(text, language),
        daemon=True
    ).start()

recognizer = sr.Recognizer()


def listen_microphone():

    global listening
    global voice_status

    if listening:
        return

    listening = True

    voice_status = "Listening..."

    try:

        with sr.Microphone() as source:

            recognizer.adjust_for_ambient_noise(
                source,
                duration=0.5
            )

            audio = recognizer.listen(
                source,
                timeout=5,
                phrase_time_limit=10
            )

        voice_status = "Recognizing..."

        if language_mode == "ta":

            text = recognizer.recognize_google(
                audio,
                language="ta-IN"
            )

        elif language_mode == "en":

            text = recognizer.recognize_google(
                audio,
                language="en-IN"
            )

        else:

            try:

                text = recognizer.recognize_google(
                    audio,
                    language="ta-IN"
                )

            except Exception:

                text = recognizer.recognize_google(
                    audio,
                    language="en-IN"
                )

        print(
            "You:",
            text
        )

        process_user_message(
            text
        )

    except sr.WaitTimeoutError:

        voice_status = "No speech detected"

    except sr.UnknownValueError:

        voice_status = "Could not understand"

    except sr.RequestError:

        voice_status = (
            "Speech service unavailable"
        )

    except Exception as error:

        print(
            "Microphone error:",
            error
        )

        voice_status = "Microphone error"

    time.sleep(1)

    voice_status = "Ready"

    listening = False


def start_microphone():

    if not listening:

        threading.Thread(
            target=listen_microphone,
            daemon=True
        ).start()

def process_user_message(text):

    global message

    if not text.strip():
        return

    language = detect_language(
        text
    )

    conversation.append(
        (
            "You",
            text
        )
    )

    response = generate_pet_reply(
        text
    )

    conversation.append(
        (
            pet_name,
            response
        )
    )

    message = response

    print(
        "Pet:",
        response
    )

    speak_async(
        response,
        language
    )

def load_pet_image(path):

    global pet_image
    global pet_image_path

    try:

        image = Image.open(
            path
        ).convert(
            "RGBA"
        )

        image.thumbnail(
            (350, 350),
            Image.Resampling.LANCZOS
        )

        pet_image = pygame.image.fromstring(
            image.tobytes(),
            image.size,
            "RGBA"
        )

        pet_image_path = path

        return True

    except Exception as error:

        print(
            "Image error:",
            error
        )

        return False


def load_pet_thumbnails():

    global pet_thumbnails

    pet_thumbnails = []

    for name, path in PET_FILES:

        thumbnail = None

        if os.path.exists(path):

            try:

                image = Image.open(
                    path
                ).convert(
                    "RGBA"
                )

                image.thumbnail(
                    (85, 85),
                    Image.Resampling.LANCZOS
                )

                thumbnail = pygame.image.fromstring(
                    image.tobytes(),
                    image.size,
                    "RGBA"
                )

            except Exception as error:

                print(
                    "Thumbnail error:",
                    error
                )

        pet_thumbnails.append(
            (
                name,
                path,
                thumbnail
            )
        )

# BUTTON CLASS
# ============================================================

class Button:

    def __init__(
        self,
        rect,
        text,
        color,
        function
    ):

        self.rect = pygame.Rect(
            rect
        )

        self.text = text
        self.color = color
        self.function = function

    def draw(self):

        color = self.color

        if self.rect.collidepoint(
            pygame.mouse.get_pos()
        ):

            color = tuple(
                min(
                    255,
                    c + 20
                )
                for c in color
            )

        pygame.draw.rect(
            screen,
            color,
            self.rect,
            border_radius=12
        )

        text = font_tiny.render(
            self.text,
            True,
            BLACK
        )

        screen.blit(
            text,
            text.get_rect(
                center=self.rect.center
            )
        )

    def click(self, position):

        if self.rect.collidepoint(
            position
        ):

            self.function()

def draw_pet():

    global talk_frame

    if pet_image is None:
        return

    x = 500
    y = 350

    # Jump

    if jump_amount > 0:

        y -= jump_amount

    # Dance

    if dance_timer > 0:

        x += int(
            15 *
            pygame.math.Vector2(
                1,
                0
            ).rotate(
                dance_timer * 10
            ).x
        )

    pet_rect = pet_image.get_rect(
        center=(x, y)
    )

    screen.blit(
        pet_image,
        pet_rect
    )

    # ========================================================
    # TALKING ANIMATION
    # ========================================================

    if speaking:

        talk_frame += 1

        # Mouth moves between two sizes

        if (
            talk_frame // 8
        ) % 2 == 0:

            mouth_height = 6

        else:

            mouth_height = 15

        mouth_width = 28

        mouth_x = x - (
            mouth_width // 2
        )

        mouth_y = y + 60

        pygame.draw.ellipse(
            screen,
            (45, 25, 25),
            (
                mouth_x,
                mouth_y,
                mouth_width,
                mouth_height
            )
        )

    else:

        # Small normal mouth

        pygame.draw.arc(
            screen,
            (60, 30, 30),
            (
                x - 12,
                y + 55,
                24,
                15
            ),
            0,
            3.14,
            2
        )

last_need_update = time.time()


def update_needs():

    global hunger
    global happiness
    global energy
    global health
    global last_need_update
    global mood

    now = time.time()

    if now - last_need_update >= 60:

        hunger = max(
            0,
            hunger - 1
        )

        happiness = max(
            0,
            happiness - 1
        )

        energy = max(
            0,
            energy - 1
        )

        if hunger < 20:

            mood = "Hungry"

        elif energy < 20:

            mood = "Tired"

        elif happiness < 20:

            mood = "Sad"

        else:

            mood = "Happy"

        last_need_update = now

while running:

    # EVENTS
    # ========================================================

        if event.type == pygame.QUIT:

            running = False

        # ====================================================
        # CREATE PET SCREEN
        # ====================================================

        if not game_started:

            if event.type == pygame.MOUSEBUTTONDOWN:

                (
                    pet_rects,
                    own_rect,
                    name_rect,
                    personality_rect,
                    action_rects,
                    start_rect
                ) = draw_start_screen()

                mouse = event.pos

                # Select pet

                for index, rect in pet_rects:

                    if rect.collidepoint(
                        mouse
                    ):

                        select_builtin_pet(
                            index
                        )

                # Own image

                if own_rect.collidepoint(
                    mouse
                ):

                    choose_own_image()

                # Name

                if name_rect.collidepoint(
                    mouse
                ):

                    name_box_active = True

                else:

                    name_box_active = False

                # Personality

                if personality_rect.collidepoint(
                    mouse
                ):

                    personality_index = (
                        personality_index + 1
                    ) % len(
                        personality_options
                    )

                # Actions

                for action, rect in action_rects:

                    if rect.collidepoint(
                        mouse
                    ):

                        if action in selected_actions:

                            selected_actions.remove(
                                action
                            )

                        else:

                            selected_actions.append(
                                action
                            )

                # Start

                if start_rect.collidepoint(
                    mouse
                ):

                    start_game()

            # Name typing

            if event.type == pygame.KEYDOWN:

                if name_box_active:

                    if event.key == pygame.K_BACKSPACE:

                        typing_name = (
                            typing_name[:-1]
                        )

                    elif event.key == pygame.K_RETURN:

                        name_box_active = False

                    else:

                        if len(typing_name) < 20:

                            if event.unicode.isprintable():

                                typing_name += (
                                    event.unicode
                                )

        # ====================================================
        # GAME
        # ====================================================

        else:

            if event.type == pygame.MOUSEBUTTONDOWN:

                mouse = event.pos

                    )

                        language_mode = "auto"
